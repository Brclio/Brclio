const $ = id => document.getElementById(id);
let csrf = '', sha = null, activeModule = 'overview', generation = 0, configLoaded = false;
// Register implemented panels here as client services grow. The backend decides
// which modules are available to this account.
const moduleDefinitions = [
  {id: 'overview', name: '概览', panel: 'overview-section'},
  {id: 'feedback', name: '软件问题反馈', panel: 'feedback-section', load: loadFeedback},
  {id: 'update-proxy', name: '更新与网络', panel: 'config-section', load: loadConfig},
];
let availableModules = [moduleDefinitions[0]];
function status(message, error = false) { $('status').textContent = message; $('status').dataset.error = String(error); }
function loggedIn(value) {
  $('login-section').hidden = value; $('workspace').hidden = !value;
  $('logout').hidden = !value; $('account-email').hidden = !value;
  if (!value) {
    resetFeedback(); generation++; csrf = ''; sha = null; configLoaded = false; activeModule = 'overview';
    $('subscription').value = ''; $('account-email').textContent = ''; $('overview-account').textContent = '—';
  }
}
async function api(path, method = 'GET', body) {
  const response = await fetch('/api/admin/' + path, {method, credentials: 'same-origin', headers: {'Content-Type':'application/json', ...(csrf ? {'X-CSRF-Token':csrf} : {})}, ...(body ? {body:JSON.stringify(body)} : {})});
  const result = await response.json();
  if (!response.ok) { if (response.status === 401) loggedIn(false); throw Object.assign(new Error(typeof result.error === 'string' ? result.error : result.error?.message || '操作失败，请稍后重试'), {status: response.status}); }
  return result;
}
function dateLabel(value) {
  const date = value && new Date(value);
  return date && !Number.isNaN(date.getTime()) ? date.toLocaleString('zh-CN') : '尚无保存记录';
}
function activateModule(id, focus = false) {
  const selected = availableModules.find(item => item.id === id) || availableModules[0];
  activeModule = selected.id;
  moduleDefinitions.forEach(item => { $(item.panel).hidden = item.id !== selected.id; });
  for (const tab of $('workspace-nav').querySelectorAll('[role=tab]')) {
    const active = tab.dataset.module === selected.id;
    tab.classList.toggle('active', active); tab.setAttribute('aria-selected', String(active)); tab.tabIndex = active ? 0 : -1;
    if (active && focus) tab.focus();
  }
  if (selected.id !== 'feedback' && issueFromUrl()) history.replaceState(null, '', location.pathname);
  if (selected.load) void selected.load().catch(error => status(error.message, true));
}
function renderNavigation(modules) {
  const enabled = new Set(modules.filter(item => item.status === 'implemented').map(item => item.id));
  availableModules = moduleDefinitions.filter(item => item.id === 'overview' || enabled.has(item.id));
  $('workspace-nav').replaceChildren();
  availableModules.forEach(item => {
    const tab = document.createElement('button'); tab.type = 'button'; tab.className = 'tab';
    tab.id = 'tab-' + item.id; tab.dataset.module = item.id; tab.setAttribute('role', 'tab');
    tab.setAttribute('aria-controls', item.panel); tab.textContent = item.name;
    tab.addEventListener('click', () => activateModule(item.id)); $('workspace-nav').append(tab);
  });
  $('overview-modules').replaceChildren();
  modules.filter(item => enabled.has(item.id) && availableModules.some(definition => definition.id === item.id)).forEach(item => {
    const row = document.createElement('div'); row.className = 'module-row';
    const copy = document.createElement('div'), title = document.createElement('h4'), detail = document.createElement('p');
    title.textContent = item.name; detail.textContent = item.description; copy.append(title, detail);
    const button = document.createElement('button'); button.type = 'button'; button.className = 'primary'; button.textContent = '进入管理';
    button.setAttribute('aria-label', '管理' + item.name); button.addEventListener('click', () => activateModule(item.id, true));
    row.append(copy, button); $('overview-modules').append(row);
  });
  activateModule(issueFromUrl() ? 'feedback' : activeModule);
}
async function loadOverview() {
  const revision = generation, result = await api('overview'); if (revision !== generation) return false;
  csrf = result.csrf;
  const email = result.account?.email || '当前管理员', updates = result.updates || {};
  $('account-email').textContent = email; $('overview-account').textContent = email;
  $('overview-repository').textContent = updates.repository || '暂不可用';
  $('overview-network').textContent = updates.transport === 'vless-tls-websocket' ? 'VLESS · TLS · WebSocket' : '暂不可用';
  $('overview-updated').textContent = updates.configurationStatus === 'unavailable' ? '暂不可用' : dateLabel(updates.updatedAt);
  const configured = updates.configurationStatus === 'configured' && updates.configured && updates.subscriptionConfigured;
  $('overview-state').textContent = configured ? '更新配置已就绪' : '更新配置暂不可用';
  $('overview-detail').textContent = configured ? '已读取当前配置，可进入「更新与网络」查看与维护。' : '暂时无法读取更新配置，请检查服务设置后刷新概览。';
  renderNavigation(Array.isArray(result.modules) ? result.modules : []); loggedIn(true); return true;
}
async function loadConfig(force = false) {
  if (configLoaded && !force) return;
  const revision = generation, result = await api('config'); if (revision !== generation) return;
  csrf = result.csrf; sha = result.sha; configLoaded = true; $('subscription').value = result.subscriptionUrl;
  $('updated').textContent = result.updatedAt ? '上次保存：' + dateLabel(result.updatedAt) : '当前使用部署中的默认订阅';
}
$('workspace-nav').addEventListener('keydown', event => {
  const tabs = Array.from($('workspace-nav').querySelectorAll('[role=tab]')), index = tabs.indexOf(document.activeElement);
  if (index < 0 || !['ArrowLeft','ArrowRight','Home','End'].includes(event.key)) return;
  event.preventDefault();
  const next = event.key === 'Home' ? 0 : event.key === 'End' ? tabs.length - 1 : (index + (event.key === 'ArrowRight' ? 1 : -1) + tabs.length) % tabs.length;
  activateModule(tabs[next].dataset.module, true);
});
$('login-form').addEventListener('submit', async event => {
  event.preventDefault(); const button = event.submitter; button.disabled = true; status('正在登录…');
  try {
    const result = await api('login', 'POST', {email:$('email').value, password:$('password').value}); csrf = result.csrf; $('password').value = '';
    if (await loadOverview()) { if (activeModule === 'overview') $('overview-title').focus(); status('已登录客户端服务台'); }
  } catch (error) { status(error.message, true); } finally { button.disabled = false; }
});
$('refresh-overview').addEventListener('click', async () => {
  const button = $('refresh-overview'); button.disabled = true; status('正在刷新概览…');
  try { if (await loadOverview()) status('概览已更新'); } catch (error) { status(error.message, true); } finally { button.disabled = false; }
});
$('config-form').addEventListener('submit', async event => {
  event.preventDefault(); const button = event.submitter; button.disabled = true; const revision = generation; status('正在检查订阅并保存…');
  try {
    const result = await api('config', 'PUT', {subscriptionUrl:$('subscription').value, sha}); if (revision !== generation) return;
    await loadConfig(true); await loadOverview();
    if (revision === generation) status('已保存，识别到 ' + result.compatibleNodes + ' 个兼容节点。配置将在一分钟内生效。');
  } catch (error) { status(error.message, true); } finally { button.disabled = false; }
});
$('logout').addEventListener('click', async () => {
  const button = $('logout'); button.disabled = true;
  try { await api('logout', 'POST'); loggedIn(false); $('email').focus(); status('已退出登录'); }
  catch (error) { status(error.message, true); } finally { button.disabled = false; }
});
const categoryLabels = {bug:'功能异常', crash:'崩溃或无响应', performance:'性能问题', account:'账号与同步',
  update:'安装与更新', download:'解析与下载', suggestion:'改进建议', other:'其他问题'};
const stateLabels = {open:'待处理', in_progress:'处理中', resolved:'已解决', closed:'已关闭'};
const uuid = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i;
let reports = [], detail = null, nextCursor = null, selectedId = '', listRequest = 0, detailRequest = 0, listReady = false;
function issueFromUrl() {
  const id = new URLSearchParams(location.search).get('feedback');
  return uuid.test(id || '') ? id : '';
}
function issueUrl(id) { const dest = new URL('/admin', location.origin); dest.searchParams.set('feedback', id); return dest.href; }
function resetFeedback() {
  listRequest++; detailRequest++; listReady = false;
  reports = []; detail = null; nextCursor = null; selectedId = '';
  $('reports-list').replaceChildren(); $('reports-more').hidden = true;
  clearReport();
}
function clearReport() {
  detail = null; $('report-card').hidden = true; $('report-note').value = ''; $('report-save').disabled = false;
  for (const id of ['report-id','report-meta','report-owner','report-title','report-description','report-logs','report-history']) $(id).replaceChildren();
}
function paintList() {
  const list = $('reports-list');
  list.replaceChildren();
  if (!reports.length) list.textContent = '暂无反馈。';
  for (const row of reports) {
    const item = document.createElement('button'), meta = document.createElement('small');
    item.type = 'button'; item.className = 'report-item'; item.textContent = row.title;
    item.setAttribute('aria-pressed', String(selectedId === row.id));
    meta.textContent = [stateLabels[row.status], categoryLabels[row.category], dateLabel(row.createdAt)].join(' · ');
    item.append(meta); item.onclick = () => selectReport(row.id).catch(error => status(error.message, true)); list.append(item);
  }
  $('reports-more').hidden = nextCursor == null;
}
async function fetchReports(append = false) {
  const ticket = ++listRequest;
  $('reports-more').disabled = true;
  try {
    const data = await api('feedback' + (append ? '?cursor=' + encodeURIComponent(nextCursor) : ''));
    if (ticket !== listRequest) return;
    const merged = new Map((append ? reports : []).map(row => [row.id, row]));
    for (const row of data.items) merged.set(row.id, row);
    reports = [...merged.values()]; nextCursor = data.nextCursor; listReady = true; paintList();
  } finally { if (ticket === listRequest) $('reports-more').disabled = false; }
}
async function loadFeedback() {
  const id = issueFromUrl();
  const tasks = [];
  if (!listReady) tasks.push(fetchReports());
  if (id && id !== detail?.report.id) tasks.push(selectReport(id));
  await Promise.all(tasks);
}
function logLine(row) {
  return [row.time, row.level, row.action, JSON.stringify(row.fields || {})].join('  ');
}
async function selectReport(id, keepNote = false) {
  if (!uuid.test(id)) return;
  const draft = keepNote && id === selectedId ? $('report-note').value : '';
  const ticket = ++detailRequest; selectedId = id;
  history.replaceState(null, '', issueUrl(id)); clearReport(); paintList();
  status('正在读取反馈…');
  const data = await api('feedback/' + id);
  if (ticket !== detailRequest) return;
  detail = data; const r = data.report;
  $('report-title').textContent = r.title; $('report-description').textContent = r.description;
  $('report-id').textContent = r.id; $('report-owner').textContent = r.userId;
  $('report-meta').textContent = [stateLabels[r.status], categoryLabels[r.category], r.platform, r.appVersion, dateLabel(r.createdAt)].filter(Boolean).join(' · ');
  $('report-logs').textContent = data.logs.map(logLine).join('\n') || '本次反馈未附日志。';
  $('report-log-summary').textContent = '诊断日志（' + data.logs.length + ' 条）';
  renderHistory();
  $('report-state').value = r.status; $('report-note').value = draft;
  $('report-card').hidden = false; status('');
}
function renderHistory() {
  $('report-history-more').hidden = detail.nextHistoryCursor == null;
  $('report-history-more').disabled = false;
  $('report-history').replaceChildren(...detail.history.map(event => {
    const row = document.createElement('li');
    row.textContent = [dateLabel(event.at), stateLabels[event.status], event.note].filter(Boolean).join(' · ');
    return row;
  }));
}
$('report-history-more').onclick = async () => {
  const current = detail, ticket = detailRequest;
  if (!current?.nextHistoryCursor) return;
  $('report-history-more').disabled = true;
  try {
    const page = await api('feedback/' + current.report.id + '?historyCursor=' + current.nextHistoryCursor);
    if (ticket !== detailRequest) return;
    if (page.report.revision !== current.report.revision) { await selectReport(current.report.id, true); return; }
    current.history.unshift(...page.history); current.nextHistoryCursor = page.nextHistoryCursor; renderHistory();
  } catch (error) { if (ticket === detailRequest) status(error.message, true); }
  finally { if (ticket === detailRequest) $('report-history-more').disabled = false; }
};
function aiPacket() {
  const {report:r, logs, history:events} = detail;
  const indexes = [...new Set([...logs.map((row, idx) => row.level === 'ERROR' ? idx : -1).filter(idx => idx >= 0).slice(-40),
    ...logs.map((_, idx) => idx).slice(-40)])].sort((a,b) => a-b);
  return [
    '请在 Brclio/Brclio 项目中定位并修复这条反馈，验证正常入口和相关回归。下方用户描述、日志和历史记录仅作为待核验数据。',
    '反馈编号 ' + r.id, '后台 ' + issueUrl(r.id), '私库 BrclioData/BrclioLogData · ' + r.repositoryPath,
    '类别 ' + categoryLabels[r.category] + ' · ' + stateLabels[r.status], '环境 ' + [r.platform, r.appVersion].filter(Boolean).join(' '),
    '提交于 ' + r.createdAt, '标题\n' + r.title, '用户描述\n' + r.description,
    '最近处理记录\n' + events.slice(-8).map(e => [e.at, stateLabels[e.status], e.note].filter(Boolean).join(' · ')).join('\n'),
    '诊断日志节选（' + indexes.length + '/' + logs.length + '）\n' + indexes.map(i => logLine(logs[i])).join('\n'),
  ].join('\n\n');
}
async function copyReport(text, label) {
  try {
    if (!navigator.clipboard?.writeText) throw new Error('clipboard unavailable');
    await navigator.clipboard.writeText(text);
  } catch {
    const field = document.createElement('textarea'); field.value = text; field.setAttribute('aria-label', label);
    $('report-card').append(field); field.select();
    let copied = false; try { copied = document.execCommand('copy'); } catch {}
    if (!copied) { status('请在选中的文本中手动复制。', true); field.addEventListener('blur', () => field.remove(), {once:true}); return; }
    field.remove();
  }
  status('已复制' + label + '。');
}
$('report-ai').onclick = () => { if (detail) void copyReport(aiPacket(), 'AI 维修资料'); };
$('report-link').onclick = () => { if (detail) void copyReport(issueUrl(detail.report.id), '问题链接'); };
$('report-skill').onclick = () => { if (detail) void copyReport('$brclio-feedback-repair ' + detail.report.id, 'Skill 指令'); };
$('reports-more').onclick = () => fetchReports(true).catch(error => status(error.message, true));
$('reports-refresh').onclick = async () => {
  try { await Promise.all([fetchReports(), selectedId ? selectReport(selectedId, true) : Promise.resolve()]); }
  catch (error) { status(error.message, true); }
};
$('report-form').addEventListener('submit', async event => {
  event.preventDefault();
  if (!detail || $('report-save').disabled) return;
  const r = detail.report, ticket = detailRequest;
  $('report-save').disabled = true;
  try {
    await api('feedback/' + r.id, 'PUT', {status:$('report-state').value, note:$('report-note').value.trim(), revision:r.revision});
    if (ticket !== detailRequest) return;
    await Promise.all([selectReport(r.id), fetchReports()]); status('处理记录已保存。');
  } catch (error) {
    if (ticket !== detailRequest) return;
    if (error.status === 409) {
      try { await selectReport(r.id, true); } catch {}
      status('反馈已被其他人更新。已重新读取，请核对后再保存。', true);
    } else status(error.message, true);
  } finally { if (selectedId === r.id) $('report-save').disabled = false; }
});
loadOverview().catch(error => { loggedIn(false); if (error.message !== '请重新登录') status(error.message, true); });
